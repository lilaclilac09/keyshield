1//! Store key instruction handler
2
3use pinocchio::{
4    account_info::AccountInfo,
5    program_error::ProgramError,
6    pubkey::Pubkey,
7    ProgramResult,
8};
9
10use crate::{
11    error::KeyShieldError,
12    pda::derive_vault_pda,
13    state::Vault,
14};
15
16/// Process store key instruction
17/// 
18/// Accounts:
19/// 0. [signer] Owner - The wallet storing the key
20/// 1. [writable] Vault - PDA account to store the encrypted key
21/// 2. [] System Program
22pub fn process_store_key(
23    program_id: &Pubkey,
24    accounts: &[AccountInfo],
25    data: &[u8],
26) -> ProgramResult {
27    // Expected data: encrypted_key (128) + zk_commit (32) + mpc_hash (32) + timestamp (8) = 200 bytes
28    if data.len() < 200 {
29        return Err(KeyShieldError::InvalidKeyData.into());
30    }
31
32    let accounts_iter = &mut accounts.iter();
33    let owner = accounts_iter.next().ok_or(ProgramError::NotEnoughAccountKeys)?;
34    let vault = accounts_iter.next().ok_or(ProgramError::NotEnoughAccountKeys)?;
35
36    // Verify owner is signer
37    if !owner.is_signer {
38        return Err(KeyShieldError::InvalidVaultOwner.into());
39    }
40
41    // Verify vault PDA
42    let (expected_vault_pda, _bump) = derive_vault_pda(program_id, owner.key)?;
43    if *vault.key != expected_vault_pda {
44        return Err(KeyShieldError::InvalidVaultOwner.into());
45    }
46
47    // Check if vault already exists (skip if account is being initialized)
48    let vault_data_len = vault.data.borrow().len();
49    if vault_data_len > 0 && vault_data_len >= Vault::SIZE {
50        // Check discriminator to see if it's already initialized
51        let existing_discriminator = &vault.data.borrow()[0..8];
52        if existing_discriminator == Vault::DISCRIMINATOR {
53            return Err(KeyShieldError::VaultAlreadyExists.into());
54        }
55    }
56
57    // Parse instruction data
58    let mut encrypted_key = [0u8; 128];
59    encrypted_key.copy_from_slice(&data[0..128]);
60    
61    let mut zk_commit = [0u8; 32];
62    zk_commit.copy_from_slice(&data[128..160]);
63    
64    let mut mpc_hash = [0u8; 32];
65    mpc_hash.copy_from_slice(&data[160..192]);
66    
67    let timestamp = u64::from_le_bytes(
68        data[192..200].try_into().map_err(|_| KeyShieldError::InvalidKeyData)?
69    );
70
71    // Create vault state
72    let vault_state = Vault::new(
73        *owner.key,
74        encrypted_key,
75        zk_commit,
76        mpc_hash,
77        timestamp,
78    );
79
80    // Serialize and write to account
81    let mut vault_data = vault.data.borrow_mut();
82    if vault_data.len() < Vault::SIZE {
83        return Err(ProgramError::AccountDataTooSmall);
84    }
85
86    // Write discriminator
87    vault_data[0..8].copy_from_slice(&vault_state.discriminator);
88    // Write owner
89    vault_data[8..40].copy_from_slice(vault_state.owner.as_ref());
90    // Write encrypted_key
91    vault_data[40..168].copy_from_slice(&vault_state.encrypted_key);
92    // Write zk_commit
93    vault_data[168..200].copy_from_slice(&vault_state.zk_commit);
94    // Write mpc_hash
95    vault_data[200..232].copy_from_slice(&vault_state.mpc_hash);
96    // Write created_at
97    vault_data[232..240].copy_from_slice(&vault_state.created_at.to_le_bytes());
98    // Write access_flags
99    vault_data[240] = vault_state.access_flags;
100    // Reserved bytes remain zero
101
102    Ok(())
103}
