35/// Program entrypoint
36fn process_instruction(
37    program_id: &Pubkey,
38    accounts: &[AccountInfo],
39    instruction_data: &[u8],
40) -> ProgramResult {
41    // Instruction data must have at least 1 byte for discriminator
42    if instruction_data.is_empty() {
43        return Err(ProgramError::InvalidInstructionData);
44    }
45
46    // Parse instruction discriminator (first byte)
47    let instruction = Instruction::try_from_u8(instruction_data[0])
48        .ok_or(ProgramError::InvalidInstructionData)?;
49
50    // Route to instruction handler (skip discriminator byte)
51    let data = &instruction_data[1..];
52
53    match instruction {
54        Instruction::StoreKey => process_store_key(program_id, accounts, data),
55        Instruction::AccessKey => process_access_key(program_id, accounts, data),
56        Instruction::ShareKey => process_share_key(program_id, accounts, data),
57    }
58}
